import React from 'react'; import { AbsoluteFill, Composition, Easing, interpolate, useCurrentFrame } from 'remotion'; import { RumahUtuh as Rumah, BAGIAN } from './gambar/Rumah.gen'; import { P5Kanvas, useP5 } from './gambar/P5';
const io = Easing.inOut(Easing.cubic), masuk = (f: number, a: number) => interpolate(f, [a, a + 60], [0, 1], { easing: io, extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
const UjiRumah: React.FC = () => { const f = useCurrentFrame(); return <AbsoluteFill style={{ background: '#FDFDFD' }}><Rumah lebar={1000} style={{ left: 460, top: 100 }} gaya={Object.fromEntries(BAGIAN.map((b, i) => { const g = masuk(f, i * 20); return [b, b === 'atap' ? { transform: `translateY(${-300 * (1 - g)}px)`, opacity: g } : b === 'pintu' || b === 'jendela' ? { transform: `scale(${g})` } : { opacity: g }]; }))} /></AbsoluteFill>; };
const UjiGelombang: React.FC = () => { const f = useCurrentFrame(), p = useP5(7); const G = Array.from({ length: 5 }, (_, k) => Array.from({ length: 97 }, (_, i) => `${i ? 'L' : 'M'}${i * 20} ${(540 + (k - 2) * 70 + (p.noise(i * 0.06, k * 3, f * 0.012) - 0.5) * 320).toFixed(2)}`).join('')); const T = Array.from({ length: 40 }, () => [p.random(1920), p.random(1080), p.random(4, 12)]);
  return <AbsoluteFill style={{ background: '#FDFDFD' }}><svg width={1920} height={1080}>{T.map(([x, y, r], i) => <circle key={i} cx={x} cy={y} r={r} fill="#E49C2C" opacity={0.5} />)}{G.map((d, i) => <path key={i} d={d} fill="none" stroke="#5F4DEE" strokeWidth={4} opacity={0.35 + i * 0.15} />)}</svg></AbsoluteFill>; };
const lukis = (p: any, t: { frame: number }) => { p.noStroke(); p.fill('#5F4DEE'); for (let i = 0; i < 60; i++) { const a = i / 60 * p.TWO_PI + t.frame * 0.01, r = 250 + p.noise(i * 0.2, t.frame * 0.02) * 220; p.circle(960 + Math.cos(a) * r, 540 + Math.sin(a) * r, 8 + p.random(20)); } };
const UjiKanvas: React.FC = () => <AbsoluteFill style={{ background: '#FDFDFD' }}><P5Kanvas width={1920} height={1080} seed={3} gambar={lukis} /></AbsoluteFill>;
export const Root: React.FC = () => <>{([['Rumah', UjiRumah], ['Gelombang', UjiGelombang], ['Kanvas', UjiKanvas]] as const).map(([id, c]) => <Composition key={id} id={id} component={c} durationInFrames={180} fps={60} width={1920} height={1080} />)}</>;
